# Mapping Display Unit Type to Unit Types

<https://jeremytammik.github.io/tbc/a/1051_displayunittype.htm>

```csharp
/// &lt;summary&gt; /// Map each DisplayUnitType to a list of all the /// UnitType values that it might be used for, e.g. /// Meters is mapped to the following 21 values: /// Length, SheetLength, HVAC_DuctSize, HVAC_Roughness, /// PipeSize, Piping_Roughness, WireSize, DecSheetLength, /// Electrical_CableTraySize, Electrical_ConduitSize, /// Reinforcement_Length, HVAC_DuctInsulationThickness, /// HVAC_DuctLiningThickness, PipeInsulationThickness, /// Bar_Diameter, Crack_Width, Displacement_Deflection, /// Reinforcement_Cover, Reinforcement_Spacing, /// Section_Dimension, Section_Property. /// &lt;/summary&gt; class MapDutToUt : Dictionary&lt;DisplayUnitType, List&lt;UnitType&gt;&gt; { &nbsp; public MapDutToUt() &nbsp; { &nbsp; &nbsp; IList&lt;DisplayUnitType&gt; duts; &nbsp; &nbsp; &nbsp; Array a = Enum.GetValues( typeof( UnitType ) ); &nbsp; &nbsp; &nbsp; foreach( UnitType ut in a ) &nbsp; &nbsp; { &nbsp; &nbsp; &nbsp; // Skip the UT_Undefined and UT_Custom entries; &nbsp; &nbsp; &nbsp; // GetValidDisplayUnits throws ArgumentException &nbsp; &nbsp; &nbsp; // on them, saying &quot;unitType is an invalid unit &nbsp; &nbsp; &nbsp; // type.&nbsp; See UnitUtils.IsValidUnitType() and &nbsp; &nbsp; &nbsp; // UnitUtils.GetValidUnitTypes().&quot; &nbsp; &nbsp; &nbsp; &nbsp; if( UnitType.UT_Undefined == ut &nbsp; &nbsp; &nbsp; &nbsp; || UnitType.UT_Custom == ut ) &nbsp; &nbsp; &nbsp; { &nbsp; &nbsp; &nbsp; &nbsp; continue; &nbsp; &nbsp; &nbsp; } &nbsp; &nbsp; &nbsp; &nbsp; duts = UnitUtils.GetValidDisplayUnits( ut ); &nbsp; &nbsp; &nbsp; &nbsp; foreach( DisplayUnitType dut in duts ) &nbsp; &nbsp; &nbsp; { &nbsp; &nbsp; &nbsp; &nbsp; //Debug.Assert( !ContainsKey( dut ), &nbsp; &nbsp; &nbsp; &nbsp; //&nbsp; &quot;unexpected duplicate DisplayUnitType key&quot; ); &nbsp; &nbsp; &nbsp; &nbsp; &nbsp; if( !ContainsKey( dut ) ) &nbsp; &nbsp; &nbsp; &nbsp; { &nbsp; &nbsp; &nbsp; &nbsp; &nbsp; Add( dut, new List&lt;UnitType&gt;( 1 ) ); &nbsp; &nbsp; &nbsp; &nbsp; } &nbsp; &nbsp; &nbsp; &nbsp; this[dut].Add( ut ); &nbsp; &nbsp; &nbsp; } &nbsp; &nbsp; } &nbsp; } }
```
