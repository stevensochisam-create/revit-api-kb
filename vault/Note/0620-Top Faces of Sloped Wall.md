public static class JtEdgeArrayExtensionMethods { &nbsp; /// &lt;summary&gt; &nbsp; /// Return a polygon as a list of XYZ points from &nbsp; /// an EdgeArray. If any of the edges are curved, &nbsp; /// we retrieve the tessellated points, i.e. an &nbsp; /// approximation determined by Revit. &nbsp; /// &lt;/summary&gt; &nbsp; public static List&lt;XYZ&gt; GetPolygon( &nbsp; &nbsp; this EdgeArray ea ) &nbsp; { &nbsp; &nbsp; int n = ea.Size; &nbsp; &nbsp; &nbsp; List&lt;XYZ&gt; polygon = new List&lt;XYZ&gt;( n ); &nbsp; &nbsp; &nbsp; foreach( Edge e in ea ) &nbsp; &nbsp; { &nbsp; &nbsp; &nbsp; IList&lt;XYZ&gt; pts = e.Tessellate(); &nbsp; &nbsp; &nbsp; &nbsp; n = polygon.Count; &nbsp; &nbsp; &nbsp; &nbsp; if( 0 &lt; n ) &nbsp; &nbsp; &nbsp; { &nbsp; &nbsp; &nbsp; &nbsp; Debug.Assert( pts[0] &nbsp; &nbsp; &nbsp; &nbsp; &nbsp; .IsAlmostEqualTo( polygon[n-1] ), &nbsp; &nbsp; &nbsp; &nbsp; &nbsp; &quot;expected last edge end point to &quot; &nbsp; &nbsp; &nbsp; &nbsp; &nbsp; + &quot;equal next edge start point&quot; ); &nbsp; &nbsp; &nbsp; &nbsp; &nbsp; polygon.RemoveAt( n - 1 ); &nbsp; &nbsp; &nbsp; } &nbsp; &nbsp; &nbsp; polygon.AddRange( pts ); &nbsp; &nbsp; } &nbsp; &nbsp; n = polygon.Count; &nbsp; &nbsp; &nbsp; Debug.Assert( polygon[0] &nbsp; &nbsp; &nbsp; .IsAlmostEqualTo( polygon[n - 1] ), &nbsp; &nbsp; &nbsp; &quot;expected first edge start point to &quot; &nbsp; &nbsp; &nbsp; + &quot;equal last edge end point&quot; ); &nbsp; &nbsp; &nbsp; polygon.RemoveAt( n - 1 ); &nbsp; &nbsp; &nbsp; return polygon; &nbsp; } }
