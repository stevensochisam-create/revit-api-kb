/// &lt;summary&gt; /// A bounding box for a collection of XYZ instances. /// The components of a tuple are read-only and cannot /// be changed after instantiation, so I cannot use /// that easily. /// The components of an XYZ are read-only and cannot /// be changed except by re-instantiation, so I cannot /// use that easily either. /// &lt;/summary&gt; class JtBoundingBoxXyz // : Tuple&lt;XYZ, XYZ&gt; { &nbsp; /// &lt;summary&gt; &nbsp; /// Minimum and maximum X, Y and Z values. &nbsp; /// &lt;/summary&gt; &nbsp; double xmin, ymin, zmin, xmax, ymax, zmax; &nbsp; &nbsp; /// &lt;summary&gt; &nbsp; /// Initialise to infinite values. &nbsp; /// &lt;/summary&gt; &nbsp; public JtBoundingBoxXyz() &nbsp; { &nbsp; &nbsp; xmin = ymin = zmin = double.MaxValue; &nbsp; &nbsp; xmax = ymax = zmax = double.MinValue; &nbsp; } &nbsp; &nbsp; /// &lt;summary&gt; &nbsp; /// Return current lower left corner. &nbsp; /// &lt;/summary&gt; &nbsp; public XYZ Min &nbsp; { &nbsp; &nbsp; get { return new XYZ( xmin, ymin, zmin ); } &nbsp; } &nbsp; &nbsp; /// &lt;summary&gt; &nbsp; /// Return current upper right corner. &nbsp; /// &lt;/summary&gt; &nbsp; public XYZ Max &nbsp; { &nbsp; &nbsp; get { return new XYZ( xmax, ymax, zmax ); } &nbsp; } &nbsp; &nbsp; public XYZ MidPoint &nbsp; { &nbsp; &nbsp; get { return 0.5 * ( Min + Max ); } &nbsp; } &nbsp; &nbsp; /// &lt;summary&gt; &nbsp; /// Expand bounding box to contain &nbsp; /// the given new point. &nbsp; /// &lt;/summary&gt; &nbsp; public void ExpandToContain( XYZ p ) &nbsp; { &nbsp; &nbsp; if( p.X &lt; xmin ) { xmin = p.X; } &nbsp; &nbsp; if( p.Y &lt; ymin ) { ymin = p.Y; } &nbsp; &nbsp; if( p.Z &lt; zmin ) { zmin = p.Z; } &nbsp; &nbsp; if( p.X &gt; xmax ) { xmax = p.X; } &nbsp; &nbsp; if( p.Y &gt; ymax ) { ymax = p.Y; } &nbsp; &nbsp; if( p.Z &gt; zmax ) { zmax = p.Z; } &nbsp; } }

&nbsp; public Point2dInt Min &nbsp; { &nbsp; &nbsp; get { return new Point2dInt( xmin, ymin ); } &nbsp; } &nbsp; &nbsp; /// &lt;summary&gt; &nbsp; /// Return current upper right corner. &nbsp; /// &lt;/summary&gt; &nbsp; public Point2dInt Max &nbsp; { &nbsp; &nbsp; get { return new Point2dInt( xmax, ymax ); } &nbsp; } &nbsp; &nbsp; /// &lt;summary&gt; &nbsp; /// Return current center point. &nbsp; /// &lt;/summary&gt; &nbsp; (int)(0.5 * ( ymin + ymax )) ); &nbsp; &nbsp; } &nbsp; } &nbsp; &nbsp; /// &lt;summary&gt; &nbsp; /// Expand bounding box to contain &nbsp; /// the given new point. &nbsp; /// &lt;/summary&gt;
