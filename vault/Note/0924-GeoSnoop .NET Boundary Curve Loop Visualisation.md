Note that a closing segment &nbsp; /// to connect the last point back to the first &nbsp; /// is added. &nbsp; /// &lt;/summary&gt; &nbsp; public List&lt;Point[]&gt; GetGraphicsPathLines() &nbsp; { &nbsp; &nbsp; int i, n; &nbsp; &nbsp; &nbsp; List&lt;Point[]&gt; loops &nbsp; &nbsp; &nbsp; = new List&lt;Point[]&gt;( Count ); &nbsp; &nbsp; &nbsp; foreach( JtLoop jloop in this ) &nbsp; &nbsp; { &nbsp; &nbsp; &nbsp; n = jloop.Count; &nbsp; &nbsp; &nbsp; Point[] loop = new Point[n + 1]; &nbsp; &nbsp; &nbsp; i = 0; &nbsp; &nbsp; &nbsp; foreach( Point2dInt p in jloop ) &nbsp; &nbsp; &nbsp; { &nbsp; &nbsp; &nbsp; &nbsp; loop[i++] = new Point( p.X, p.Y ); &nbsp; &nbsp; &nbsp; } &nbsp; &nbsp; &nbsp; loop[i] = loop[0]; &nbsp; &nbsp; &nbsp; loops.Add( loop ); &nbsp; &nbsp; } &nbsp; &nbsp; return loops; &nbsp; } }
